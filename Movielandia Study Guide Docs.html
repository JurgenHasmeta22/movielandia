getMoviesWithFilters — Pattern

"use server";
import { prisma } from "../../prisma/config/prisma";

export async function getMoviesWithFilters(
  { sortBy, ascOrDesc, perPage = 12, page = 1, title, filterValue,
    filterNameString, filterOperatorString }: MovieModelParams,
  userId?: number,
) {
  const skip = (page - 1) * perPage;
  const filters: any = {};

  if (title) filters.title = { contains: title };

  if (filterValue && filterNameString && filterOperatorString) {
    const operator = filterOperatorString === ">" ? "gt"
                   : filterOperatorString === "<" ? "lt" : "equals";
    filters[filterNameString] = { [operator]: filterValue };
  }

  const movies = await prisma.movie.findMany({
    where: filters,
    orderBy: { [sortBy || "title"]: ascOrDesc || "asc" },
    skip, take: perPage,
  });
  // … aggregate ratings, merge, return
}

Review Upsert Action

"use server";
import { revalidatePath } from "next/cache";

export async function upsertMovieReview(
  movieId: number, userId: number,
  content: string, rating: number
) {
  await prisma.movieReview.upsert({
    where:  { userId_movieId: { userId, movieId } },
    update: { content, rating, updatedAt: new Date() },
    create: { userId, movieId, content, rating },
  });
  revalidatePath(`/movies/${movieId}`); // bust Next.js cache for the page
}

Upvote / Downvote Action Pattern

export async function addUpvoteMovieReview(
  userId: number, movieId: number, movieReviewId: number
) {
  const existing = await prisma.upvoteMovieReview.findFirst({
    where: { userId, movieId, movieReviewId }
  });
  if (existing) {
    await prisma.upvoteMovieReview.delete({ where: { id: existing.id } });
  } else {
    await prisma.upvoteMovieReview.create({ data: { userId, movieId, movieReviewId } });
    // also remove any existing downvote for this user/review
    await prisma.downvoteMovieReview.deleteMany({ where: { userId, movieReviewId } });
  }
  revalidatePath(`/movies/${movieId}`);
}

// app/api/search/route.ts
import { NextRequest, NextResponse } from "next/server";
import { getServerSession } from "next-auth";

export async function GET(req: NextRequest) {
  const term = req.nextUrl.searchParams.get("term") ?? "";
  if (!term) return NextResponse.json([]);

  const results = await prisma.movie.findMany({
    where: { title: { contains: term, mode: "insensitive" } },
    take: 5,
    select: { id: true, title: true, photoSrc: true },
  });
  return NextResponse.json(results);
}

tryCatch — Async Error Wrapper

// src/utils/helpers/tryCatch.ts
export async function tryCatch<T, E = Error>(
  promise: Promise<T>
): Promise<Result<T, E>> {
  try {
    const data = await promise;
    return { data, error: null };
  } catch (error) {
    return { data: null, error: error as E };
  }
}

// Usage:
const { data, error } = await tryCatch(getMovieById(id));
if (error) return notFound();

useDebounce — Implementation

"use client";
import { useState, useEffect } from "react";

export function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const timer = setTimeout(() => setDebouncedValue(value), delay);
    return () => clearTimeout(timer);
  }, [value, delay]);

  return debouncedValue;
}

// Usage in search component:
const [query, setQuery] = useState("");
const debouncedQuery = useDebounce(query, 300);

useEffect(() => {
  if (debouncedQuery) fetchResults(debouncedQuery);
}, [debouncedQuery]);

Prisma Query Patterns

Pagination + Sorting (standard pattern)

const skip = (page - 1) * perPage;

const [items, total] = await prisma.$transaction([
  prisma.movie.findMany({
    where: filters,
    orderBy: { [sortBy]: ascOrDesc },
    skip,
    take: perPage,
    include: { genres: true, reviews: true },
  }),
  prisma.movie.count({ where: filters }),
]);

Aggregate Ratings (groupBy)

const ratings = await prisma.movieReview.groupBy({
  by: ["movieId"],
  where: { movieId: { in: movieIds } },
  _avg: { rating: true },
  _count: { rating: true },
});

// Build a lookup map for O(1) merge:
const ratingsMap = ratings.reduce((acc, r) => {
  acc[r.movieId] = {
    averageRating: r._avg.rating ?? 0,
    totalReviews:  r._count.rating,
  };
  return acc;
}, {} as RatingsMap);

// Merge into movie objects:
const moviesWithRatings = movies.map((m) => ({
  ...m,
  ...ratingsMap[m.id],
}));

Include Relations (deep select)

const movie = await prisma.movie.findUnique({
  where: { id: movieId },
  include: {
    genres: { include: { genre: true } },
    cast:   { include: { actor: true }, take: 10 },
    crew:   { include: { crew:  true }, take: 10 },
    reviews: {
      include: {
        user: { select: { userName: true, photoSrc: true } },
        upvotes: true,
        downvotes: true,
      },
      orderBy: { createdAt: "desc" },
      take: 10,
    },
  },
});

Upsert Pattern

// Favorite toggle — add if absent, else remove
const existing = await prisma.userMovieFavorite.findUnique({
  where: { userId_movieId: { userId, movieId } }
});

if (existing) {
  await prisma.userMovieFavorite.delete({ where: { id: existing.id } });
} else {
  await prisma.userMovieFavorite.create({ data: { userId, movieId } });
}

NextAuth Configuration

Auth lives at src/app/api/auth/[...nextauth]/route.ts. Uses next-auth v4 with @next-auth/prisma-adapter.

authOptions Structure

import { PrismaAdapter } from "@next-auth/prisma-adapter";
import CredentialsProvider from "next-auth/providers/credentials";
import { compareSync } from "bcrypt";

export const authOptions: NextAuthOptions = {
  adapter: PrismaAdapter(prisma),
  providers: [
    CredentialsProvider({
      name: "credentials",
      credentials: {
        email:    { label: "Email",    type: "email"    },
        password: { label: "Password", type: "password" },
      },
      async authorize(credentials) {
        const user = await prisma.user.findUnique({
          where: { email: credentials?.email }
        });
        if (!user || !user.password) return null;
        if (!user.active) throw new Error("Email not verified");
        const valid = compareSync(credentials!.password, user.password);
        return valid ? user : null;
      },
    }),
  ],
  session:  { strategy: "jwt" },
  callbacks: {
    async jwt({ token, user }) {
      if (user) {
        token.id       = user.id;
        token.role     = user.role;
        token.userName = user.userName;
      }
      return token;
    },
    async session({ session, token }) {
      session.user.id       = token.id as string;
      session.user.role     = token.role as string;
      session.user.userName = token.userName as string;
      return session;
    },
  },
  pages: {
    signIn:  "/auth/signin",
    error:   "/auth/error",
  },
};

export default async function AdminLayout({ children }) {
  const session = await getServerSession(authOptions);

  if (!session?.user)         redirect("/auth/signin");
  if (session.user.role !== "Admin") redirect("/");

  return <AdminLayout>{children}</AdminLayout>;
}
